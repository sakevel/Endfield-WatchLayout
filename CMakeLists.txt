cmake_minimum_required(VERSION 3.25)
project(EndfieldWatchLayout VERSION 0.3.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(WatchLayout SHARED src/plugin.cpp)
target_include_directories(WatchLayout PRIVATE src sdk/include)
set_target_properties(WatchLayout PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/watch-layout")
add_custom_target(package ALL DEPENDS WatchLayout)
foreach(f mod.ini config.ini watch-layout.lua config-entry.lua icon.png zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/${f}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/watch-layout/${f}")
endforeach()
enable_testing()
add_executable(ContractTests tests/contract_tests.cpp)
target_include_directories(ContractTests PRIVATE src sdk/include)
add_test(NAME contract COMMAND ContractTests "$<TARGET_FILE:WatchLayout>")
set(ZML_LUPA_PATH "" CACHE PATH "Existing optional Lupa installation")
set(ZML_SERVICES_TEST_EXE "" CACHE FILEPATH "Public loader black-box fixture")
if(ZML_LUPA_PATH AND ZML_SERVICES_TEST_EXE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/behavior_tests.py"
    --lupa-dir "${ZML_LUPA_PATH}" --services "${ZML_SERVICES_TEST_EXE}")
  set_tests_properties(behavior PROPERTIES TIMEOUT 40)
endif()
